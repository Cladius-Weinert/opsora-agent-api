Quick Start

3 baris kode untuk mulai

from openai import OpenAI client = OpenAI( base_url="https://api.opsora.id/v1", api_key="opsora-sk-xxxxx", ) response = client.chat.completions.create( model="opsora-brain", messages=[{"role": "user", "content": "Halo!"}] ) print(response.choices[0].message.content)
import OpenAI from 'openai'; const client = new OpenAI({ baseURL: 'https://api.opsora.id/v1', apiKey: 'opsora-sk-xxxxx', }); const response = await client.chat.completions.create({ model: 'opsora-brain', messages: [{ role: 'user', content: 'Halo!' }], }); console.log(response.choices[0].message.content);
curl https://api.opsora.id/v1/chat/completions \ -H "Authorization: Bearer opsora-sk-xxxxx" \ -H "Content-Type: application/json" \ -d '{ "model": "opsora-brain", "messages": [ {"role": "user", "content": "Halo!"} ] }'
